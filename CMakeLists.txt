cmake_minimum_required(VERSION 3.24)
project(ThanLongAutoPK VERSION 0.1.7 LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "ThanLongAutoPK is Windows-only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ThanLongAutoPK must be built x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(ThanLongCleanRouteBridge SHARED vendor/r3/src/bridge.cpp)
target_include_directories(ThanLongCleanRouteBridge PRIVATE vendor/r3/src)
target_compile_definitions(ThanLongCleanRouteBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ThanLongCleanRouteBridge PRIVATE user32 kernel32)
set_target_properties(ThanLongCleanRouteBridge PROPERTIES OUTPUT_NAME "ThanLongCleanRouteBridge")

add_executable(ThanLongAutoPK WIN32 src/controller.cpp)
target_include_directories(ThanLongAutoPK PRIVATE vendor/r3/src)
target_compile_definitions(ThanLongAutoPK PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ThanLongAutoPK PRIVATE comctl32 user32 kernel32 gdi32)
target_link_options(ThanLongAutoPK PRIVATE /MANIFEST:NO)

if(MSVC)
  target_compile_options(ThanLongCleanRouteBridge PRIVATE /utf-8 /W4 /permissive- /FIstdlib.h)
  target_compile_options(ThanLongAutoPK PRIVATE /utf-8 /W4 /permissive-)
endif()
